# Blender 5.2 Batch Bone Renaming & Retargeting Utility

# Automatically normalizes Mixamo armature naming to Rigify / Humanoid standards

import bpy



def normalize_armature_bones(armature_obj, target_standard="RIGIFY"):

    """

    Renames bones from mixamorig: namespace to target standard (RIGIFY or HUMANOID).

    Preserves parent relationships, edit bone roll vectors, and bone collections.

    """

    if armature_obj.type != 'ARMATURE':

        print(f"Error: {armature_obj.name} is not an Armature.")

        return False

    

    # Mapping Table: Mixamo -> (Rigify, Humanoid)

    BONE_MAP = {

        # Spine & Head

        "mixamorig:Hips": ("spine", "Hips"),

        "mixamorig:Spine": ("spine.001", "Spine"),

        "mixamorig:Spine1": ("spine.002", "Chest"),

        "mixamorig:Spine2": ("spine.003", "UpperChest"),

        "mixamorig:Neck": ("neck", "Neck"),

        "mixamorig:Head": ("head", "Head"),

        

        # Left Arm

        "mixamorig:LeftShoulder": ("shoulder.L", "LeftShoulder"),

        "mixamorig:LeftArm": ("upper_arm.L", "LeftUpperArm"),

        "mixamorig:LeftForeArm": ("forearm.L", "LeftLowerArm"),

        "mixamorig:LeftHand": ("hand.L", "LeftHand"),

        

        # Right Arm

        "mixamorig:RightShoulder": ("shoulder.R", "RightShoulder"),

        "mixamorig:RightArm": ("upper_arm.R", "RightUpperArm"),

        "mixamorig:RightForeArm": ("forearm.R", "RightLowerArm"),

        "mixamorig:RightHand": ("hand.R", "RightHand"),

        

        # Left Leg

        "mixamorig:LeftUpLeg": ("thigh.L", "LeftUpperLeg"),

        "mixamorig:LeftLeg": ("shin.L", "LeftLowerLeg"),

        "mixamorig:LeftFoot": ("foot.L", "LeftFoot"),

        "mixamorig:LeftToeBase": ("toe.L", "LeftToes"),

        

        # Right Leg

        "mixamorig:RightUpLeg": ("thigh.R", "RightUpperLeg"),

        "mixamorig:RightLeg": ("shin.R", "RightLowerLeg"),

        "mixamorig:RightFoot": ("foot.R", "RightFoot"),

        "mixamorig:RightToeBase": ("toe.R", "RightToes"),

        

        # Hand Fingers (Left)

        "mixamorig:LeftHandThumb1": ("thumb.01.L", "LeftThumbProximal"),

        "mixamorig:LeftHandThumb2": ("thumb.02.L", "LeftThumbIntermediate"),

        "mixamorig:LeftHandThumb3": ("thumb.03.L", "LeftThumbDistal"),

        "mixamorig:LeftHandIndex1": ("f_index.01.L", "LeftIndexProximal"),

        "mixamorig:LeftHandIndex2": ("f_index.02.L", "LeftIndexIntermediate"),

        "mixamorig:LeftHandIndex3": ("f_index.03.L", "LeftIndexDistal"),

        "mixamorig:LeftHandMiddle1": ("f_middle.01.L", "LeftMiddleProximal"),

        "mixamorig:LeftHandMiddle2": ("f_middle.02.L", "LeftMiddleIntermediate"),

        "mixamorig:LeftHandMiddle3": ("f_middle.03.L", "LeftMiddleDistal"),

        "mixamorig:LeftHandRing1": ("f_ring.01.L", "LeftRingProximal"),

        "mixamorig:LeftHandRing2": ("f_ring.02.L", "LeftRingIntermediate"),

        "mixamorig:LeftHandRing3": ("f_ring.03.L", "LeftRingDistal"),

        "mixamorig:LeftHandPinky1": ("f_pinky.01.L", "LeftLittleProximal"),

        "mixamorig:LeftHandPinky2": ("f_pinky.02.L", "LeftLittleIntermediate"),

        "mixamorig:LeftHandPinky3": ("f_pinky.03.L", "LeftLittleDistal"),

    }

    

    target_idx = 0 if target_standard.upper() == "RIGIFY" else 1

    renamed_count = 0

    

    bpy.context.view_layer.objects.active = armature_obj

    bpy.ops.object.mode_set(mode='EDIT')

    

    for edit_bone in armature_obj.data.edit_bones:

        orig_name = edit_bone.name

        clean_name = orig_name.strip()

        

        # Handle prefix strip or direct dictionary lookup

        if clean_name in BONE_MAP:

            new_name = BONE_MAP[clean_name][target_idx]

            edit_bone.name = new_name

            renamed_count += 1

        elif clean_name.startswith("mixamorig:"):

            # Strip namespace prefix if not in table

            suffix = clean_name.replace("mixamorig:", "")

            edit_bone.name = suffix

            renamed_count += 1

            

    bpy.ops.object.mode_set(mode='OBJECT')

    print(f"Renamed {renamed_count} bones to {target_standard} standard.")

    return True



# Execute on active object in Blender 5.2

if bpy.context.active_object and bpy.context.active_object.type == 'ARMATURE':

    normalize_armature_bones(bpy.context.active_object, target_standard="RIGIFY")

else:

    print("Select an Armature object before executing this script.")
